    def _validate_cloud_supported(self) -> "McpGatewayEventConsumerConfig":
        """Refuse to boot on a cloud this service has no lane for.

        The Azure branch this service used to carry was a working-looking dead path: the Service Bus
        body is an Event Grid event, the S3-notification parser returned no records for it, and the
        handler acked — healthy pod, zero errors, 100% of messages dropped. The branch is gone; this
        makes its absence a startup refusal instead of a silent drop, and says plainly that the Azure
        gateway lane does not exist yet — rather than pointing an operator at the SCANNER lane, which is a
        different object source and is not wired for gateway objects on Azure.
        """
        assert_cloud_supported(
            self.cloud,
            service="mcp_gateway_event_consumer_service",
            supported=CloudProvider.AWS,
            alternative=(
                "There is no Azure implementation of the MCP-GATEWAY object lane yet: the Azure side ships one "
                "Service Bus queue, endpoint-asset-ingestion, fed by a sys.Label 'source=mcp-scanner' rule, so "
                "gateway objects are not routed on an Azure cell at all. endpoint_asset_ingestion CAN run its "
                "gateway lane (ONYX__CLOUD=azure), but it would need its own Service Bus queue and an Event Grid "
                "subscription rule for source=mcp-gateway, neither of which exists. Run this service on an AWS "
                "cell, or track the Azure gateway lane before enabling one."
            ),
        )
        return self
